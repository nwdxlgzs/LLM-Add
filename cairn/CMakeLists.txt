cmake_minimum_required(VERSION 3.20)
project(cairn VERSION 0.1.0 LANGUAGES C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
option(CAIRN_OPENMP "Use OpenMP for large matrix rows" ON)
option(CAIRN_SANITIZE "Address/undefined sanitizers (GCC/Clang)" OFF)
option(CAIRN_SVE "Build isolated SVE kernels when the compiler supports them" ON)
option(CAIRN_RVV "Build isolated RVV 1.0 kernels when the compiler supports them" ON)
if(CAIRN_NATIVE)
  message(WARNING "CAIRN_NATIVE is obsolete and ignored: runtime dispatch builds portable ISA variants")
endif()
add_library(cairn STATIC src/cairn.c src/quant.c src/tokenizer.c src/cpu.c src/cpu_x86.c src/cpu_neon.c src/cpu_wasm.c)
target_include_directories(cairn PUBLIC include)
include(CheckCSourceCompiles)
if(CMAKE_SYSTEM_PROCESSOR MATCHES "^(aarch64|arm64|ARM64)$" AND NOT MSVC AND CAIRN_SVE)
  set(CMAKE_REQUIRED_FLAGS "-march=armv8-a+sve")
  check_c_source_compiles("#include <arm_sve.h>\nint main(void){return (int)svcntw();}" CN_COMPILER_SVE)
  unset(CMAKE_REQUIRED_FLAGS)
  if(CN_COMPILER_SVE)
    target_sources(cairn PRIVATE src/cpu_sve.c)
    set_source_files_properties(src/cpu_sve.c PROPERTIES COMPILE_OPTIONS "-march=armv8-a+sve")
    target_compile_definitions(cairn PRIVATE CN_BUILD_SVE=1)
  endif()
endif()
if(CMAKE_SYSTEM_PROCESSOR MATCHES "riscv64" AND CAIRN_RVV)
  set(CMAKE_REQUIRED_FLAGS "-march=rv64gcv")
  check_c_source_compiles("#include <riscv_vector.h>\nint main(void){return (int)__riscv_vsetvl_e32m1(4);}" CN_COMPILER_RVV)
  unset(CMAKE_REQUIRED_FLAGS)
  if(CN_COMPILER_RVV)
    target_sources(cairn PRIVATE src/cpu_rvv.c)
    set_source_files_properties(src/cpu_rvv.c PROPERTIES COMPILE_OPTIONS "-march=rv64gcv")
    target_compile_definitions(cairn PRIVATE CN_BUILD_RVV=1)
  endif()
endif()
if(MSVC)
  target_compile_options(cairn PRIVATE /W4 /O2)
else()
  target_compile_options(cairn PRIVATE -Wall -Wextra -Wpedantic -O3)
  target_link_libraries(cairn PUBLIC m)
  # Stable baseline and scalar reference. ISA variants use function targets or
  # per-source flags; never enable -march=native for the entire library.
  if(CMAKE_C_COMPILER_ID STREQUAL "GNU")
    set_source_files_properties(src/cpu.c PROPERTIES COMPILE_OPTIONS "-fno-tree-vectorize;-ffp-contract=off")
  elseif(CMAKE_C_COMPILER_ID MATCHES "Clang")
    set_source_files_properties(src/cpu.c PROPERTIES COMPILE_OPTIONS "-fno-vectorize;-fno-slp-vectorize;-ffp-contract=off")
  endif()
  if(CAIRN_SANITIZE)
    target_compile_options(cairn PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(cairn PUBLIC -fsanitize=address,undefined)
  endif()
endif()
if(CAIRN_OPENMP)
  find_package(OpenMP)
  if(OpenMP_C_FOUND)
    target_link_libraries(cairn PUBLIC OpenMP::OpenMP_C)
  endif()
endif()
add_executable(cairn_cli src/main.c)
target_link_libraries(cairn_cli PRIVATE cairn)
set_target_properties(cairn_cli PROPERTIES OUTPUT_NAME cairn)
enable_testing()
add_executable(cairn_tests tests/test.c)
target_link_libraries(cairn_tests PRIVATE cairn)
add_test(NAME engine COMMAND cairn_tests)
add_executable(cairn_custom examples/custom.c)
target_link_libraries(cairn_custom PRIVATE cairn)
add_test(NAME custom_config COMMAND cairn_custom)
add_executable(cairn_cpu_tests tests/cpu_test.c)
target_link_libraries(cairn_cpu_tests PRIVATE cairn)
add_test(NAME cpu_backends COMMAND cairn_cpu_tests)
if(NOT MSVC)
  target_compile_options(cairn_cli PRIVATE -Wall -Wextra -Wpedantic)
  target_compile_options(cairn_tests PRIVATE -Wall -Wextra -Wpedantic)
  target_compile_options(cairn_custom PRIVATE -Wall -Wextra -Wpedantic)
  target_compile_options(cairn_cpu_tests PRIVATE -Wall -Wextra -Wpedantic)
endif()
